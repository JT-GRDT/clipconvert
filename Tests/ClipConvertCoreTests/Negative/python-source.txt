#!/usr/bin/env python3
"""
Utility helpers for batching records before they are written to disk.

This module intentionally has no external dependencies so it can be
copied into other projects without dragging in a package manager.
"""

import json
import time


# Default batch size used when the caller does not specify one.
DEFAULT_BATCH_SIZE = 100


class RecordBatcher:
    # Accumulates records until the batch is full, then flushes them.
    def __init__(self, batch_size=DEFAULT_BATCH_SIZE):
        self.batch_size = batch_size
        self._buffer = []

    def add(self, record):
        # Append a single record and flush automatically when full.
        self._buffer.append(record)
        if len(self._buffer) >= self.batch_size:
            self.flush()

    def flush(self):
        # Write whatever is currently buffered and clear the buffer.
        if not self._buffer:
            return
        payload = json.dumps(self._buffer)
        self._write(payload)
        self._buffer = []

    def _write(self, payload):
        # Placeholder write - real implementations replace this.
        timestamp = time.time()
        print(f"[{timestamp}] flushing {len(payload)} bytes")


def load_config(path):
    # Read a JSON config file and fall back to defaults on failure.
    try:
        with open(path) as handle:
            return json.load(handle)
    except FileNotFoundError:
        return {"batch_size": DEFAULT_BATCH_SIZE}


if __name__ == "__main__":
    config = load_config("batcher.json")
    batcher = RecordBatcher(batch_size=config.get("batch_size", DEFAULT_BATCH_SIZE))
    for i in range(250):
        batcher.add({"id": i, "value": i * 2})
    batcher.flush()
