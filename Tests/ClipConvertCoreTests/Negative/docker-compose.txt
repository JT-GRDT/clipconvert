# Local development stack for the sample service.
# Bring it up with: docker compose up -d

version: "3.9"

services:
  # Primary application container.
  app:
    build: .
    ports:
      - "8080:8080"
    environment:
      - APP_ENV=development
      - LOG_LEVEL=debug
    depends_on:
      - db
      - cache
    volumes:
      - ./src:/app/src

  # Relational database used for persistent storage.
  db:
    image: postgres:16
    restart: unless-stopped
    environment:
      - POSTGRES_USER=devuser
      - POSTGRES_PASSWORD=devpass
      - POSTGRES_DB=sample
    ports:
      - "5432:5432"
    volumes:
      - db-data:/var/lib/postgresql/data

  # In-memory cache for session data.
  cache:
    image: redis:7-alpine
    restart: unless-stopped
    ports:
      - "6379:6379"

# Named volumes persist data across container restarts.
volumes:
  db-data:
